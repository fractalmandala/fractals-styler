    function saveState() {
      const stateObj = {
        root: JSON.parse(JSON.stringify(root)),
        selectedId: selectedId
      };
      const stateStr = JSON.stringify(stateObj);

      if (historyIndex >= 0 && historyStack[historyIndex] && JSON.stringify(historyStack[historyIndex]) === stateStr) {
        return;
      }

      if (historyIndex < historyStack.length - 1) {
        historyStack = historyStack.slice(0, historyIndex + 1);
      }
      historyStack.push(JSON.parse(stateStr));
      historyIndex = historyStack.length - 1;
      updateUndoRedoButtons();
    }

    function undo() {
      if (historyIndex > 0) {
        historyIndex--;
        const state = historyStack[historyIndex];
        root = JSON.parse(JSON.stringify(state.root));
        selectedId = state.selectedId;
        renderCanvas(false);
        updateUndoRedoButtons();
      }
    }

    function redo() {
      if (historyIndex < historyStack.length - 1) {
        historyIndex++;
        const state = historyStack[historyIndex];
        root = JSON.parse(JSON.stringify(state.root));
        selectedId = state.selectedId;
        renderCanvas(false);
        updateUndoRedoButtons();
      }
    }

    function updateUndoRedoButtons() {
      document.getElementById('btnUndo').disabled = historyIndex <= 0;
      document.getElementById('btnRedo').disabled = historyIndex >= historyStack.length - 1;
    }

        widthVal: options.widthVal || '',

        height: options.height || 'auto',
        heightVal: options.heightVal || '',
        minHVal: options.minHVal || '',
        maxHVal: options.maxHVal || '',
        marginBot: options.marginBot !== undefined ? options.marginBot : 0,
        primitive: options.primitive || 'box',
        buttonVariant: options.buttonVariant || 'default',
        padding: options.padding !== undefined ? options.padding : 0,
        gap: options.gap !== undefined ? options.gap : 0,
        gridCols: options.gridCols || 3,
        colSpan: options.colSpan || 1,
        rowSpan: options.rowSpan || 1,
        radius: options.radius || 'radius0',

        textAlign: options.textAlign || 'left',
        borderWidth: options.borderWidth || '0',
        shadow: options.shadow || 'none',
        smDirection: options.smDirection || 'default',
        children: options.children || [],
        content: options.content || null
      };
    }

    let root = getLayoutTemplate('pancake');
    let selectedId = root.id;

    function findNode(node, id) {
      if (node.id === id) return node;
      for (const child of node.children) {
        const found = findNode(child, id);
        if (found) return found;
      }
      return null;
    }

    function findParent(node, id) {
      for (let i = 0; i < node.children.length; i++) {
        if (node.children[i].id === id) return { parent: node, index: i };
        const res = findParent(node.children[i], id);
        if (res) return res;
      }
      return null;
    }

    function getNodePath(node, id, path = []) {
      const currentPath = [...path, node];
      if (node.id === id) return currentPath;
      for (const child of node.children) {
        const res = getNodePath(child, id, currentPath);
        if (res) return res;
      }
      return null;
    }

    function renderCanvas(recordState = true) {
      const appShell = document.getElementById('appShell');
      if (isPreviewMode) {
        appShell.classList.add('mode-preview');
      } else {
        appShell.classList.remove('mode-preview');
      }

      const container = document.getElementById('canvasRoot');
      container.innerHTML = '';
      container.appendChild(renderNodeDOM(root));
      renderTreeDOM();
      renderBreadcrumbs();
      updateInspector();
      updateCodeOutput();
      if (recordState) {
        saveState();
      }
    }

    function renderBreadcrumbs() {
      const container = document.getElementById('breadcrumbTrail');
      container.innerHTML = '';
      const path = getNodePath(root, selectedId) || [root];

      path.forEach((node, idx) => {
        const span = document.createElement('span');
        span.className = `crumb-item ${idx === path.length - 1 ? 'active' : ''}`;
        span.textContent = node.name;
        span.addEventListener('click', (e) => {
          e.stopPropagation();
          selectedId = node.id;
          renderCanvas(false);
        });
        container.appendChild(span);

        if (idx < path.length - 1) {
          const sep = document.createElement('span');
          sep.className = 'crumb-sep';
          sep.textContent = '>';
          container.appendChild(sep);
        }
      });
    }

    function makeInlineEditable(textEl, node) {
      textEl.setAttribute('contenteditable', 'true');
      textEl.classList.add('inline-text-editing');
      textEl.focus();

      // Select all text
      const range = document.createRange();
      range.selectNodeContents(textEl);
      const sel = window.getSelection();
      sel.removeAllRanges();
      sel.addRange(range);

      function saveInlineText() {
        textEl.removeAttribute('contenteditable');
        textEl.classList.remove('inline-text-editing');
        node.content = textEl.textContent;
        renderCanvas();
      }

      textEl.addEventListener('blur', saveInlineText, { once: true });
      textEl.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === 'Escape') {
          e.preventDefault();
          textEl.blur();
        }
      });
    }

    function renderNodeDOM(node) {
      const isSelected = node.id === selectedId;

      if (node.primitive === 'button') {
        const btn = document.createElement('button');
        let btnCls = 'btn-preview';
        if (node.buttonVariant === 'primary') btnCls += ' btn-preview-primary';
        if (node.buttonVariant === 'quiet') btnCls += ' btn-preview-quiet';
        if (node.buttonVariant === 'icon') btnCls = 'btn-preview-icon';
        btn.className = btnCls;
        btn.textContent = node.content || 'Button';

        if (isPreviewMode) {
          btn.addEventListener('click', () => {
            alert(`Clicked button "${node.content || 'Button'}"!`);
          });
        }

        const wrapper = document.createElement('div');
        wrapper.className = `builder-node ${isSelected ? 'selected' : ''}`;
        wrapper.style.cssText = 'position: relative; display: inline-block;';

        const tag = document.createElement('div');
        tag.className = 'node-tag';
        tag.textContent = `${node.name} (button.${node.buttonVariant || 'default'})`;
        wrapper.appendChild(tag);
        wrapper.appendChild(btn);

        if (!isPreviewMode) {
          wrapper.addEventListener('click', (e) => {
            e.stopPropagation();
            selectedId = node.id;
            renderCanvas(false);
          });

          // Double click for inline text editing on button
          wrapper.addEventListener('dblclick', (e) => {
            e.stopPropagation();
            makeInlineEditable(btn, node);
          });

          wrapper.addEventListener('contextmenu', (e) => {
            e.preventDefault();
            e.stopPropagation();
            selectedId = node.id;
            renderCanvas(false);
            showContextMenu(e.clientX, e.clientY);
          });
        }

        return wrapper;
      }

      const el = document.createElement('div');
      el.className = `builder-node ${isSelected ? 'selected' : ''}`;

      let style = `padding: ${node.padding}px; gap: ${node.gap}px; margin-bottom: ${node.marginBot}px;`;
      style += ` display: ${node.display};`;
      if (node.display === 'flex' || node.display === 'inline-flex') {
        style += ` flex-direction: ${node.direction};`;
        if (node.alignItems) style += ` align-items: ${node.alignItems};`;
        if (node.justifyContent) style += ` justify-content: ${node.justifyContent};`;
      }

      const effectiveCols = (viewMode === 'mobile' && node.gridCols > 1) ? 1 : node.gridCols;
      if (node.display === 'grid') style += ` grid-template-columns: repeat(${effectiveCols}, 1fr);`;

      if (node.colSpan && node.colSpan > 1) style += ` grid-column: span ${node.colSpan};`;
      if (node.rowSpan && node.rowSpan > 1) style += ` grid-row: span ${node.rowSpan};`;

      if (node.width === '100%') style += ' width: 100%;';
      else if (node.width === '100vw') style += ' width: 100vw;';
      else if (node.width === 'fill') style += ' flex-grow: 1;';
      else if (node.width === 'custom' && node.widthVal) style += ` width: ${node.widthVal.includes('px') || node.widthVal.includes('rem') ? node.widthVal : node.widthVal + 'px'};`;
      else if (node.width === 'minmax') {
        if (node.minWVal) style += ` min-width: ${node.minWVal.includes('px') || node.minWVal.includes('rem') ? node.minWVal : node.minWVal + 'px'};`;
        if (node.maxWVal) style += ` max-width: ${node.maxWVal.includes('px') || node.maxWVal.includes('rem') ? node.maxWVal : node.maxWVal + 'px'};`;
      }

      if (node.height === '100%') style += ' height: 100%;';
      else if (node.height === '100vh') style += ' height: 100vh;';
      else if (node.height === 'fill') style += ' flex-grow: 1;';
      else if (node.height === 'custom' && node.heightVal) style += ` height: ${node.heightVal.includes('px') || node.heightVal.includes('rem') ? node.heightVal : node.heightVal + 'px'};`;
      else if (node.height === 'minmax') {
        if (node.minHVal) style += ` min-height: ${node.minHVal.includes('px') || node.minHVal.includes('rem') ? node.minHVal : node.minHVal + 'px'};`;
        if (node.maxHVal) style += ` max-height: ${node.maxHVal.includes('px') || node.maxHVal.includes('rem') ? node.maxHVal : node.maxHVal + 'px'};`;
      }

      if (node.surface === 'surface') style += ' background: var(--bg-surface);';
      else if (node.surface === 'panel') style += ' background: var(--bg-raised); border: 1px solid var(--border);';
      else if (node.surface === 'raised') style += ' background: var(--bg-hover);';
      else if (node.surface === 'subtle') style += ' background: #475569;';
      else if (node.surface === 'custom') style += ` background: ${node.customBg};`;

      if (node.borderWidth && node.borderWidth !== '0') style += ` border: ${node.borderWidth} solid ${node.borderColor || '#334155'};`;
      if (node.shadow && node.shadow !== 'none') style += ' box-shadow: 0 10px 25px rgba(0,0,0,0.4);';

      const radMap = { radius2: '2px', radius4: '4px', radius8: '8px', radius12: '12px', radius16: '16px', radiusfull: '999px' };
      if (radMap[node.radius]) style += ` border-radius: ${radMap[node.radius]};`;

      el.style.cssText = style;

      const tag = document.createElement('div');
      tag.className = 'node-tag';
      tag.textContent = `${node.name} (${node.display}${node.display === 'flex' ? '-' + node.direction : ''})`;
      el.appendChild(tag);

      if (isSelected && !isPreviewMode) {
        const handleR = document.createElement('div');
        handleR.className = 'resize-handle resize-handle-r';
        attachResizeHandler(handleR, node, 'r');
        el.appendChild(handleR);

        const handleB = document.createElement('div');
        handleB.className = 'resize-handle resize-handle-b';
        attachResizeHandler(handleB, node, 'b');
        el.appendChild(handleB);

        const handleBR = document.createElement('div');
        handleBR.className = 'resize-handle resize-handle-br';
        attachResizeHandler(handleBR, node, 'br');
        el.appendChild(handleBR);
      }

      let textNodeEl = null;
      if (node.content) {
        textNodeEl = document.createElement('div');
        textNodeEl.style.cssText = `color: ${node.textColor || 'var(--text-secondary)'}; font-weight: 600; text-align: ${node.textAlign || 'left'};`;
        textNodeEl.textContent = node.content;
        el.appendChild(textNodeEl);
      }

      for (const child of node.children) {
        el.appendChild(renderNodeDOM(child));
      }

      if (!isPreviewMode) {
        el.addEventListener('click', (e) => {
          e.stopPropagation();
          selectedId = node.id;
          renderCanvas(false);
        });

        // Double click inline text editing if text present, else select parent
        el.addEventListener('dblclick', (e) => {
          e.stopPropagation();
          if (textNodeEl) {
            makeInlineEditable(textNodeEl, node);
          } else {
            const res = findParent(root, node.id);
            if (res) {
              selectedId = res.parent.id;
              renderCanvas(false);
            }
          }
        });

        el.addEventListener('contextmenu', (e) => {
          e.preventDefault();
          e.stopPropagation();
          selectedId = node.id;
          renderCanvas(false);
          showContextMenu(e.clientX, e.clientY);
        });

        el.addEventListener('dragover', (e) => {
          e.preventDefault();
          e.stopPropagation();
          el.classList.add('drag-target');
        });

        el.addEventListener('dragleave', (e) => {
          e.preventDefault();
          e.stopPropagation();
          el.classList.remove('drag-target');
        });

        el.addEventListener('drop', (e) => {
          e.preventDefault();
          e.stopPropagation();
          el.classList.remove('drag-target');
          if (draggedType) {
            const newNode = createAddedNode(draggedType);
            node.children.push(newNode);
            selectedId = newNode.id;
            renderCanvas();
          }
        });
      }

      return el;
    }

    const dimTooltip = document.getElementById('dimTooltip');

    function attachResizeHandler(handleEl, node, direction) {
      handleEl.addEventListener('mousedown', (e) => {
        e.stopPropagation();
        e.preventDefault();

        const startX = e.clientX;
        const startY = e.clientY;

        const targetEl = handleEl.parentElement;
        const startWidth = targetEl.offsetWidth;
        const startHeight = targetEl.offsetHeight;

        const parentInfo = findParent(root, node.id);
        const isParentGrid = parentInfo && parentInfo.parent.display === 'grid';
        const gridCols = isParentGrid ? (parentInfo.parent.gridCols || 12) : 12;
        const parentW = isParentGrid ? parentInfo.parent.offsetWidth : 960;
        const colWidth = parentW / gridCols;

        dimTooltip.style.display = 'block';

        function onMouseMove(me) {
          me.preventDefault();
          const dx = me.clientX - startX;
          const dy = me.clientY - startY;

          dimTooltip.style.left = `${me.clientX + 16}px`;
          dimTooltip.style.top = `${me.clientY + 16}px`;

          let newW = Math.max(40, startWidth + dx);
          let newH = Math.max(30, startHeight + dy);

          if (isParentGrid) {
            const calculatedSpan = Math.min(gridCols, Math.max(1, Math.round(newW / colWidth)));
            node.colSpan = calculatedSpan;
            dimTooltip.textContent = `▦ Grid Span: col-span-${calculatedSpan} (${Math.round(newW)}px)`;
          } else {
            if (snapMode) {
              newW = Math.round(newW / 8) * 8;
              newH = Math.round(newH / 8) * 8;
            }

            if (direction === 'r' || direction === 'br') {
              node.width = 'custom';
              node.widthVal = `${Math.round(newW)}`;
            }

            if (direction === 'b' || direction === 'br') {
              node.height = 'custom';
              node.heightVal = `${Math.round(newH)}`;
            }

            dimTooltip.textContent = `📐 ${Math.round(targetEl.offsetWidth)}px × ${Math.round(targetEl.offsetHeight)}px`;
          }

          renderCanvas(false);
        }

        function onMouseUp() {
          dimTooltip.style.display = 'none';
          window.removeEventListener('mousemove', onMouseMove);
          window.removeEventListener('mouseup', onMouseUp);
          saveState();
        }

        window.addEventListener('mousemove', onMouseMove);
        window.addEventListener('mouseup', onMouseUp);
      });
    }

          return JSON.parse(JSON.stringify(customSavedBlocks[idx].node));
        }
      }

      return createNode(addType.includes('skeleton') || addType.includes('text') ? 'box' : addType, {
        name: addType.replace('-', '_'),
        primitive: addType.includes('skeleton') || addType.includes('text') ? 'box' : addType,
        content: addType.includes('skeleton') || addType.includes('text') ? addType : null
      });
    }

    function generateSass(node, depth) {
      const indent = '\t'.repeat(depth);
      if (node.primitive === 'button') {
        const clsMap = { default: '.button', primary: '.button-primary', quiet: '.button-quiet', icon: '.icon-button' };
        return `${indent}${clsMap[node.buttonVariant || 'default'] || '.button'}\n`;
      }
      let out = `${indent}.${node.name.toLowerCase()}\n`;
      out += `${indent}\tdisplay: ${node.display}\n`;
      if (node.display === 'flex' || node.display === 'inline-flex') out += `${indent}\tflex-direction: ${node.direction}\n`;
      if (node.alignItems && node.alignItems !== 'stretch') out += `${indent}\talign-items: ${node.alignItems}\n`;
      if (node.justifyContent && node.justifyContent !== 'flex-start') out += `${indent}\tjustify-content: ${node.justifyContent}\n`;
      if (node.display === 'grid') out += `${indent}\tgrid-template-columns: repeat(${node.gridCols}, minmax(0, 1fr))\n`;

      if (node.colSpan && node.colSpan > 1) out += `${indent}\tgrid-column: span ${node.colSpan}\n`;
      if (node.rowSpan && node.rowSpan > 1) out += `${indent}\tgrid-row: span ${node.rowSpan}\n`;

      if (node.width === '100%') out += `${indent}\twidth: 100%\n`;
      else if (node.width === '100vw') out += `${indent}\twidth: 100vw\n`;
      else if (node.width === 'fill') out += `${indent}\tflex-grow: 1\n`;
      else if (node.width === 'custom' && node.widthVal) out += `${indent}\twidth: ${node.widthVal.includes('px') || node.widthVal.includes('rem') ? node.widthVal : node.widthVal + 'px'}\n`;
      else if (node.width === 'minmax') {
        if (node.minWVal) out += `${indent}\tmin-width: ${node.minWVal.includes('px') || node.minWVal.includes('rem') ? node.minWVal : node.minWVal + 'px'}\n`;
        if (node.maxWVal) out += `${indent}\tmax-width: ${node.maxWVal.includes('px') || node.maxWVal.includes('rem') ? node.maxWVal : node.maxWVal + 'px'}\n`;
      }

      if (node.height === '100%') out += `${indent}\theight: 100%\n`;
      else if (node.height === '100vh') out += `${indent}\theight: 100vh\n`;
      else if (node.height === 'fill') out += `${indent}\tflex-grow: 1\n`;
      else if (node.height === 'custom' && node.heightVal) out += `${indent}\theight: ${node.heightVal.includes('px') || node.heightVal.includes('rem') ? node.heightVal : node.heightVal + 'px'}\n`;
      else if (node.height === 'minmax') {
        if (node.minHVal) out += `${indent}\tmin-height: ${node.minHVal.includes('px') || node.minHVal.includes('rem') ? node.minHVal : node.minHVal + 'px'}\n`;
        if (node.maxHVal) out += `${indent}\tmax-height: ${node.maxHVal.includes('px') || node.maxHVal.includes('rem') ? node.maxHVal : node.maxHVal + 'px'}\n`;
      }

      if (node.padding > 0) out += `${indent}\tpadding: ${node.padding}px\n`;
      if (node.gap > 0) out += `${indent}\tgap: ${node.gap}px\n`;
      if (node.marginBot > 0) out += `${indent}\tmargin-bottom: ${node.marginBot}px\n`;
      if (node.surface === 'custom') out += `${indent}\tbackground-color: ${node.customBg}\n`;
      else if (node.surface !== 'none') out += `${indent}\tbackground-color: var(--bg-${node.surface})\n`;

      if (node.borderWidth && node.borderWidth !== '0') out += `${indent}\tborder-width: ${node.borderWidth}\n`;
      if (node.borderColor) out += `${indent}\tborder-color: ${node.borderColor}\n`;
      if (node.textColor) out += `${indent}\tcolor: ${node.textColor}\n`;
      if (node.textAlign && node.textAlign !== 'left') out += `${indent}\ttext-align: ${node.textAlign}\n`;

      for (const child of node.children) {
        out += '\n' + generateSass(child, depth + 1);
      }
      return out;
    }

    function generateHtml(node, depth) {
      const indent = '  '.repeat(depth);
      if (node.primitive === 'button') {
        const clsMap = { default: 'button', primary: 'button-primary', quiet: 'button-quiet', icon: 'icon-button' };
        const cls = clsMap[node.buttonVariant || 'default'] || 'button';
        return `${indent}<button class="${cls}">${node.content || 'Button'}</button>\n`;
      }

      const classes = [];
      if (node.display === 'flex') {
        classes.push(node.direction === 'row' ? 'row' : 'box');
      } else if (node.display === 'grid') {
        classes.push('grid', `grid-cols-${node.gridCols}`);
      } else {
        classes.push(node.display);
      }

      if (node.colSpan && node.colSpan > 1) classes.push(`col-span-${node.colSpan}`);
      if (node.rowSpan && node.rowSpan > 1) classes.push(`row-span-${node.rowSpan}`);

      if (node.smDirection && node.smDirection !== 'default') classes.push(`sm:${node.smDirection}`);
      if (node.width === '100%') classes.push('w100');
      else if (node.width === '100vw') classes.push('w100vw');
      else if (node.width === 'fill') classes.push('grow');

      if (node.height === '100%') classes.push('h100');
      else if (node.height === '100vh') classes.push('h100vh');
      else if (node.height === 'fill') classes.push('grow');

      if (node.padding > 0) classes.push(`pad${node.padding}`);
      if (node.gap > 0) classes.push(`gap${node.gap}`);
      if (node.marginBot > 0) classes.push(`marginbot${node.marginBot}`);
      if (node.radius && node.radius !== 'radius0') classes.push(node.radius);
      if (node.surface !== 'none' && node.surface !== 'custom') classes.push(node.surface);
      if (node.fontSize) classes.push(node.fontSize);
      if (node.shadow && node.shadow !== 'none') classes.push(node.shadow);

      let out = `${indent}<div class="${classes.join(' ')}"${node.surface === 'custom' ? ` style="background:${node.customBg}"` : ''}>\n`;
      if (node.content) {
        out += `${indent}  <span>${node.content}</span>\n`;
      }
      for (const child of node.children) {
        out += generateHtml(child, depth + 1);
      }
      out += `${indent}</div>\n`;
      return out;
    }

    function generateFullSvelteComponent(targetNode) {
      const compName = targetNode.name || 'LayoutComponent';
      const htmlCode = generateHtml(targetNode, 0).trim();
      const sassCode = generateSass(targetNode, 0).trim();

      return `<script lang="ts">
  // Svelte 5 Runes Component generated by fractals-styler
  let { title = '${compName}' } = $props();
</script>

<!-- Markup -->
${htmlCode}

<!-- External Indented SASS stylesheet: ${compName.toLowerCase()}.sass -->
<!--
${sassCode}
-->
`;
    }

    document.getElementById('btnCopySvelteComp').addEventListener('click', () => {
      const target = findNode(root, selectedId) || root;
      const fullCode = generateFullSvelteComponent(target);
      navigator.clipboard.writeText(fullCode);
      alert('Full Svelte 5 Component copied to clipboard!');
    });

    function switchTab(btnId, contentId) {
      ['tabLayouts', 'tabPalette', 'tabUI', 'tabMy', 'tabTree'].forEach(id => {
        document.getElementById(id).classList.remove('active');
      });
      ['layoutsContent', 'paletteContent', 'uiBlocksContent', 'myBlocksContent', 'treeContent'].forEach(id => {
        document.getElementById(id).style.display = 'none';
      });
      document.getElementById(btnId).classList.add('active');
      document.getElementById(contentId).style.display = 'flex';
    }
